# --- Turn 1: Equip Tome and Attack ---
info 1 4 # Check Leif initial state (Should have Iron Sword equipped)

equip 1

# Expected Calculations (Leif vs Bandit):
# Leif AS (Magic) = Spd (7) - Tome Wt (4) = 3
# Bandit AS = 0 (Sleep)
# Leif does NOT double (3 < 0 + 4)

# Bandit Avoid = 0 (Sleep)
# Leif Battle Hit = 99 - 0 = 99%
# Leif Base Crit = Tome Crit (0) + Skl (6) = 6% -> Battle Crit (1st hit) = 6% (capped at 6%)
# Leif Damage = (Leif Mag (5) + Tome Mt (6)) - Bandit Mag (0) = 11
attack 5 4
